        Backend sample (Node.js + Express) 
        // POST /sms-webhook  (Twilio / Exotel / MSG91 hits this)
app.post('/sms-webhook', (req, res) => {
  const { From, Body } = req.body;
  const [crop, kg, village, price, grade] =
        Body.trim().toUpperCase().split(/\s+/);
  db.listings.upsert({
    phone: From, crop, kg: +kg, village,
    price: price ? +price : null,
    grade: grade || 'PENDING', source: 'sms'
  });
  res.type('text/xml').send(
    `<Response><Message>OK ${crop} ${kg}kg listed</Message></Response>`);
});
